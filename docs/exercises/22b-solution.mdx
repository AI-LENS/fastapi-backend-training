## Solution

```python tests/load/locustfile.py
import random
from locust import HttpUser, between, task


class CoordinatorUser(HttpUser):
    weight = 5
    wait_time = between(1, 3)

    def on_start(self):
        r = self.client.post(
            "/api/v1/auth/login",
            json={"email": "alice@example.com", "password": "alice"},
        )
        self.headers = {"Authorization": f"Bearer {r.json()['access_token']}"}

    @task(5)
    def submit_screening(self):
        self.client.post(
            "/api/v1/screenings",
            headers=self.headers,
            json={
                "site_id": "SITE-001",
                "candidate_initials": "AB",
                "age": random.randint(18, 75),
                "sex": random.choice(["M", "F"]),
                "diagnosis": "Type 2 Diabetes",
                "medications": ["metformin"],
                "is_pregnant": False, "has_liver_disease": False, "in_other_trial": False,
            },
            name="POST /screenings",
        )

    @task(2)
    def list_screenings(self):
        self.client.get("/api/v1/screenings", headers=self.headers, name="GET /screenings")


class SponsorUser(HttpUser):
    weight = 1
    wait_time = between(5, 15)   # less frequent than coordinators

    def on_start(self):
        r = self.client.post(
            "/api/v1/auth/login",
            json={"email": "admin@example.com", "password": "admin"},
        )
        self.headers = {"Authorization": f"Bearer {r.json()['access_token']}"}

    @task
    def view_dlq(self):
        self.client.get(
            "/api/v1/admin/jobs?status=dead_letter",
            headers=self.headers,
            name="GET /admin/jobs?dead_letter",
        )

    @task
    def view_audit(self):
        self.client.get(
            "/api/v1/admin/audit?limit=50",
            headers=self.headers,
            name="GET /admin/audit",
        )
```

## Run

```bash
uv run locust -f tests/load/locustfile.py --host http://localhost:8000
# Now spawn distribution: 5 coordinator : 1 sponsor (per the weights)
```

## Notes

<AccordionGroup>
  <Accordion title="`name=` parameter">
    Groups requests in the report. Without it, every unique URL counts separately — `/screenings/uuid-1` ≠ `/screenings/uuid-2`. With `name="GET /screenings/{id}"`, they're aggregated.
  </Accordion>
  <Accordion title="`weight` semantics">
    Locust spawns users probabilistically by weight. With 5:1, when the test asks for 6 users you get ~5 coordinators and ~1 sponsor. Realistic mix.
  </Accordion>
</AccordionGroup>
